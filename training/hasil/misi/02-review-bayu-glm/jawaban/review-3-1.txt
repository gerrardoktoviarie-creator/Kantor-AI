=== FILE: public/app.js ===
const menu = document.querySelector('#menu');
const message = document.querySelector('#message');
const todayTotal = document.querySelector('#today-total');
const history = document.querySelector('#history');

const rupiah = (n) => `Rp${n.toLocaleString('id-ID')}`;

function say(text, isError = false) {
  message.textContent = text;
  message.classList.toggle('message--error', isError);
}

async function loadToday() {
  const res = await fetch('/api/sales/today');
  if (!res.ok) throw new Error('Gagal muat penjualan hari ini');
  const { sales, total } = await res.json();
  todayTotal.textContent = rupiah(total);
  history.replaceChildren(...sales.map((s) => {
    const item = document.createElement('li');
    const name = document.createElement('span');
    name.textContent = s.name;
    const qty = document.createElement('span');
    qty.textContent = `${s.qty}x`;
    const total = document.createElement('span');
    total.textContent = rupiah(s.total);
    const button = document.createElement('button');
    button.type = 'button';
    button.textContent = 'Batal';
    button.addEventListener('click', () => cancel(s.id));
    item.append(name, qty, total, button);
    return item;
  }));
}

async function cancel(saleId) {
  const res = await fetch(`/api/sales/${saleId}`, { method: 'DELETE' });
  if (!res.ok) throw new Error((await res.json()).error);
  say('Penjualan dibatalin.');
  await Promise.all([loadMenu(), loadToday()]);
}

history.addEventListener('click', (event) => {
  const button = event.target.closest('[data-sale]');
  if (button) cancel(button.dataset.sale);
});

async function sell(product) {
  const res = await fetch('/api/sales', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ product_id: product.id, qty: 1 }),
  });
  if (!res.ok) throw new Error((await res.json()).error);
  say(`${product.name} terjual.`);
  await Promise.all([loadMenu(), loadToday()]);
}

async function loadMenu() {
  const res = await fetch('/api/products');
  if (!res.ok) throw new Error('Gagal muat menu');
  const products = await res.json();
  menu.replaceChildren(...products.map((product) => {
    const item = document.createElement('li');
    const button = document.createElement('button');
    button.type = 'button';
    const name = document.createElement('span');
    name.textContent = product.name;
    const price = document.createElement('span');
    price.textContent = rupiah(product.price);
    const stock = document.createElement('span');
    stock.textContent = product.low ? `Stok tipis: ${product.stock}` : `Stok ${product.stock}`;
    button.disabled = product.stock === 0;
    button.append(name, stock, price);
    button.addEventListener('click', () => sell(product));
    item.append(button);
    return item;
  }));
}

try {
  await Promise.all([loadMenu(), loadToday()]);
} catch {
  say('Gak bisa nyambung ke server.', true);
}

## Alur migrasi (ditelusuri di kepala)

- **Database versi lama (user_version < 2):** migrasi 002 jalan dulu (stock + min_stock, DEFAULT), lalu 003 jalan (category DEFAULT 'minuman', Roti Bakar jadi 'makanan'), `user_version` = 3. Aman, tidak menghapus data.
- **Database versi 2 (berisi data asli di laptop kasir):** kolom `category` baru punya `DEFAULT`, jadi baris yang sudah ada langsung dapat nilai tanpa UPDATE massal; Roti Bakar dipaksa 'makanan'. Tidak ada perubahan data penjualan/stok.
- **Database baru:** skema sudah bawa kolom category, INSERT OR IGNORE produk awal langsung punya nilai. `user_version` = 3.

## Catatan rilis (untuk non-teknis)

- **Yang berubah:** menu dipisah jadi kategori *minuman* dan *makanan*. Roti Bakar Cokelat = makanan, sisa menu = minuman. Ini belum dipakai fitur kasir saat ini (hanya penataan data + persiapan laporan).
- **Langkah update:** jalankan `node server.js` biasa; server akan naikkan migrasi otomatis. **Backup dulu** file `data.db` sebelum update.
- **Risiko:** migrasi menambah satu kolom saja, tidak mengubah data penjualan/stok yang sudah ada. Risiko rendah; rollback berarti kembalikan backup `data.db`.

## Yang belum dibenerin / perlu dicek

- Tidak ada file CSS khusus kategori di proyek; pengelompokan per kategori belum tampil di UI (hanya tersimpan di DB dan urutan query). Ini scope Joko selanjutnya, bukan bug.
- Perlu dicek tampilan langsung di HP kasir setelah server di-upgrade (sama seperti review sebelumnya).
- `package.json` tidak berubah; tidak ada dependensi baru.

Verdict: **Approved after fix** — kolom NOT NULL baru sudah punya DEFAULT, migrasi hanya menambah kolom (tidak mengubah data), dan server mengurut produk per kategori.
