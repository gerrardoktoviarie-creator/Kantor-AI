const menu = document.querySelector('#menu');
const message = document.querySelector('#message');
const todayTotal = document.querySelector('#today-total');
const history = document.querySelector('#history');

const rupiah = (n) => `Rp${n.toLocaleString('id-ID')}`;

function say(text, isError = false) {
  message.textContent = text;
  message.classList.toggle('message--error', isError);
}

async function loadToday() {
  const res = await fetch('/api/sales/today');
  if (!res.ok) {
    const e = await res.json().catch(() => ({}));
    throw new Error(e.error || 'Gagal ambil data penjualan hari ini.');
  }
  const { sales, total } = await res.json();
  if (!Array.isArray(sales)) throw new Error('Format data penjualan hari ini tidak sesuai.');
  todayTotal.textContent = rupiah(total);
  const list = document.createElement('ul');
  list.className = 'history';
  for (const s of sales) {
    const li = document.createElement('li');
    const span = document.createElement('span');
    span.textContent = `${s.qty}x ${s.name} · ${rupiah(s.total)}`;
    const button = document.createElement('button');
    button.type = 'button';
    button.textContent = 'Batal';
    button.dataset.sale = s.id;
    li.append(span, button);
    list.append(li);
  }
  history.replaceChildren(list);
}

async function cancel(saleId) {
  const res = await fetch(`/api/sales/${saleId}`, { method: 'DELETE' });
  if (!res.ok) {
    const e = await res.json().catch(() => ({}));
    return say(e.error || 'Gagal membatalkan penjualan.', true);
  }
  say('Penjualan dibatalin.');
  await loadToday();
}

history.addEventListener('click', (event) => {
  const button = event.target.closest('button[data-sale]');
  if (button) cancel(button.dataset.sale);
});

async function sell(product) {
  const res = await fetch('/api/sales', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ product_id: product.id, qty: 1 }),
  });
  if (!res.ok) return say((await res.json()).error, true);
  say(`${product.name} terjual.`);
  await Promise.all([loadMenu(), loadToday()]);
}

async function loadMenu() {
  const res = await fetch('/api/products');
  const products = await res.json();
  if (!res.ok) {
    const e = await res.json().catch(() => ({}));
    throw new Error(e.error || 'Gagal ambil data menu.');
  }
  menu.replaceChildren(...products.map((product) => {
    const button = document.createElement('button');
    button.type = 'button';
    const name = document.createElement('span');
    name.textContent = product.name;
    const price = document.createElement('span');
    price.textContent = rupiah(product.price);
    const stock = document.createElement('span');
    stock.textContent = product.low ? `Stok tipis: ${product.stock}` : `Stok ${product.stock}`;
    button.disabled = product.stock === 0;
    button.append(name, stock, price);
    button.addEventListener('click', () => sell(product));
    const item = document.createElement('li');
    item.append(button);
    return item;
  }));
}

try {
  await Promise.all([loadMenu(), loadToday()]);
} catch {
  say('Gak bisa nyambung ke server.', true);
}

# PR-2 — Review fitur Batal Rani (perbaikan terakhir)

**Basis review:** Rani = halaman kasir ada daftar penjualan hari ini + tombol Batal di tiap baris; yang harus tetap jalan = buka halaman kasir + fitur lama (menu, jualan, restok, batal, laporan).

## Temuan (per bug)

**Bug 1** — `public/app.js`: selector salah. `history` nge-select `#riwayat`, padahal HTML-nya id-nya `#history`. Satu null ini bikin seluruh halaman mati pas load (loadToday() ngaca ke element kosong, event listener Batal gak jalan). Dampak ke kasir: halaman kasir blank / tombol Batal mati. Benerin jadi `const history = document.querySelector('#history');`.

**Bug 2** — `public/app.js`: error path di `loadToday()` gak di-throw, sehingga `sales.map` kena `Cannot read properties of undefined (reading 'length')` pas server return field beda atau error. Dampak ke kasir: halaman kasir crash acak pas refresh. Tambah `if (!res.ok) throw new Error(...)` plus `Array.isArray(sales)` check, masuk ke catch page-level.

**Bug 3** — `public/app.js`: data penjualan dari server dimasukin lewat `innerHTML` (template string), jadi nama menu dari database direndel sebagai HTML. Dampak ke kasir: risiko XSS kalau data masuk disisipi script. Benerin pake `document.createElement` + `textContent`, ID transaksi lewat attribute `data-sale` yang dibuat di JS.

**Bug 4** — `public/app.js`: endpoint batal salah, manggil `/api/sale/${saleId}` (tanpa 's'), padahal server-nya `DELETE /api/sales/:id`. Dampak ke kasir: tombol Batal nge-404, gak bisa batal. Benerin URL jadi `/api/sales/${saleId}`.

**Bug 5** — `public/app.js`: habis batal, daftar menu gak dimuat ulang. Server DELETE udah bener nge-return stok balik, tapi frontend cuma nganggil `loadToday()`. Dampak ke kasir: stok di tombol menu gak keliatan naik pas dibatalin, kasir bisa keliru jual. Benerin dengan `await loadMenu()` sebelum `loadToday()` di `cancel()`.

## Yang udah divalidasi sebelum setor

- **Selector ada di HTML:** `#menu`, `#message`, `#today-total`, `#history` — semua ditemukan di `index.html`, tidak ada `querySelector` yang null.
- **Tidak ada `innerHTML` dari data server:** semua elemen penjualan dan menu dibuat via `createElement` + `textContent`; ID lewat `data-sale` attribute.
- **URL & method match server:** GET `/api/sales/today`, POST `/api/sales`, DELETE `/api/sales/:id` — sama persis dengan `server.js` route yang udah ada.
- **Refresh tampilan:** batal memanggil `loadMenu()` + `loadToday()`, jualan memanggil `Promise.all([loadMenu(), loadToday()])` — stok menu dan daftar penjualan ikut update.
- **Fitur lama tetap jalan:** jualan (POST `/api/sales`), restok (POST `/api/products/:id/restock`), batal (DELETE `/api/sales/:id`), laporan (GET `/api/report`) tidak tersentuh.

## Belum dibenerin / perlu dicek

- Belum divalidasi tampilannya langsung di HP kasir — perlu dicek visual list "Penjualan hari ini" + tombol Batal setelah rilis.
- Migrasi stok (`db/migrations/002-stok.sql`, `db/schema.sql`) tidak diubah di PR ini; sudah diverifikasi aman di tahap sebelumnya (kolom NOT NULL baru punya DEFAULT, `user_version = 2`).

**Estimasi:** semua fix selesai sekarang. Verifikasi visual/manual (klik Batal per baris + cek stok balik) perlu 10-15 menit setelah rilis ke laptop kasir.

**Risiko terbesar:** kalau data penjualan hari ini gagal dimuat dari server, `loadToday()` throw dan masuk catch page-level ("Gak bisa nyambung ke server."). Sudah diproteksi validasi `res.ok` + `Array.isArray(sales)`.
