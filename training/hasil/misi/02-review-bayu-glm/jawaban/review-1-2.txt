import { createServer } from 'node:http';
import { readFileSync } from 'node:fs';
import { readFile } from 'node:fs/promises';
import { extname, join, normalize, sep } from 'node:path';
import { fileURLToPath } from 'node:url';
import { DatabaseSync } from 'node:sqlite';

const ROOT = fileURLToPath(new URL('.', import.meta.url));
const PUBLIC = join(ROOT, 'public');

const db = new DatabaseSync(process.env.DB_PATH || join(ROOT, 'data.db'));
const fresh = !db.prepare("SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'products'").get();
if (fresh) db.exec(readFileSync(join(ROOT, 'db', 'schema.sql'), 'utf8'));
else if (db.prepare('PRAGMA user_version').get().user_version < 2) {
  db.exec(readFileSync(join(ROOT, 'db', 'migrations', '002-stok.sql'), 'utf8'));
}
db.exec('PRAGMA foreign_keys = ON');

const TYPES = { '.html': 'text/html; charset=utf-8', '.css': 'text/css', '.js': 'text/javascript' };
const PRODUCT = 'SELECT id, name, price, stock, min_stock, stock <= min_stock AS low FROM products';
const toProduct = (p) => p && { ...p, low: Boolean(p.low) };

function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify(data));
}

async function readJson(req) {
  let raw = '';
  for await (const chunk of req) raw += chunk;
  try { return raw ? JSON.parse(raw) : {}; } catch { return null; }
}

function isValidDate(str) {
  if (typeof str !== 'string') return false;
  const dateRegex = /^\d{4}-\d{2}-\d{2}$/;
  if (!dateRegex.test(str)) return false;
  const [year, month, day] = str.split('-').map((v) => parseInt(v, 10));
  const d = new Date(Date.UTC(year, month - 1, day));
  return d.getUTCFullYear() === year && d.getUTCMonth() === month - 1 && d.getUTCDate() === day;
}

async function serveStatic(pathname, res) {
  const file = normalize(join(PUBLIC, pathname === '/' ? 'index.html' : pathname));
  if (!file.startsWith(PUBLIC + sep)) return sendJson(res, 404, { error: 'Gak ketemu.' });
  try {
    const body = await readFile(file);
    res.writeHead(200, { 'Content-Type': TYPES[extname(file)] ?? 'application/octet-stream' });
    res.end(body);
  } catch {
    sendJson(res, 404, { error: 'Gak ketemu.' });
  }
}

const server = createServer(async (req, res) => {
  const { pathname } = new URL(req.url, 'http://localhost');
  try {
    if (req.method === 'GET' && pathname === '/api/products') {
      return sendJson(res, 200, db.prepare(`${PRODUCT} ORDER BY name`).all().map(toProduct));
    }
    const restock = pathname.match(/^\/api\/products\/(\d+)\/restock$/);
    if (req.method === 'POST' && restock) {
      const body = await readJson(req);
      const qty = body?.qty;
      if (!Number.isInteger(qty) || qty < 1 || qty > 1000) return sendJson(res, 400, { error: 'qty harus 1-1000.' });
      const { changes } = db.prepare('UPDATE products SET stock = stock + ? WHERE id = ?').run(qty, Number(restock[1]));
      if (!changes) return sendJson(res, 404, { error: 'Menu gak ketemu.' });
      return sendJson(res, 200, toProduct(db.prepare(`${PRODUCT} WHERE id = ?`).get(Number(restock[1]))));
    }
    if (req.method === 'POST' && pathname === '/api/sales') {
      const body = await readJson(req);
      if (!body) return sendJson(res, 400, { error: 'Body harus JSON.' });
      const { product_id: productId, qty } = body;
      if (!Number.isInteger(qty) || qty < 1 || qty > 99) return sendJson(res, 400, { error: 'Jumlah harus 1-99.' });
      const product = db.prepare('SELECT id, price, stock FROM products WHERE id = ?').get(productId ?? 0);
      if (!product) return sendJson(res, 404, { error: 'Menu gak ketemu.' });
      db.exec('BEGIN');
      try {
        const { changes } = db.prepare('UPDATE products SET stock = stock - ? WHERE id = ? AND stock >= ?').run(qty, product.id, qty);
        if (!changes) {
          db.exec('ROLLBACK');
          return sendJson(res, 409, { error: `Stok gak cukup, sisa ${product.stock}.` });
        }
        const total = product.price * qty;
        const { lastInsertRowid } = db.prepare('INSERT INTO sales (product_id, qty, total) VALUES (?, ?, ?)').run(product.id, qty, total);
        db.exec('COMMIT');
        return sendJson(res, 201, { id: Number(lastInsertRowid), product_id: product.id, qty, total });
      } catch (err) {
        db.exec('ROLLBACK');
        throw err;
      }
    }
    const cancel = pathname.match(/^\/api\/sales\/(\d+)$/);
    if (req.method === 'DELETE' && cancel) {
      const sale = db.prepare(`SELECT id, product_id, qty, date(created_at, '+7 hours') = date('now', '+7 hours') AS today
        FROM sales WHERE id = ?`).get(Number(cancel[1]));
      if (!sale) return sendJson(res, 404, { error: 'Penjualan gak ketemu.' });
      if (!sale.today) return sendJson(res, 409, { error: 'Cuma penjualan hari ini yang bisa dibatalin.' });
      db.exec('BEGIN');
      db.prepare('DELETE FROM sales WHERE id = ?').run(sale.id);
      db.prepare('UPDATE products SET stock = stock + ? WHERE id = ?').run(sale.qty, sale.product_id);
      db.exec('COMMIT');
      return sendJson(res, 200, { ok: true });
    }
    if (req.method === 'GET' && pathname === '/api/sales/today') {
      const sales = db.prepare(`SELECT s.id, s.product_id, p.name, s.qty, s.total, s.created_at
        FROM sales s JOIN products p ON p.id = s.id
        WHERE date(s.created_at, '+7 hours') = date('now', '+7 hours') ORDER BY s.id DESC`).all();
      return sendJson(res, 200, { sales, grand_total: sales.reduce((sum, s) => sum + s.total, 0) });
    }
    // Laporan per menu. from/to: YYYY-MM-DD (WIB), dua-duanya ikut dihitung.
    if (req.method === 'GET' && pathname === '/api/report') {
      const params = new URL(req.url, 'http://localhost').searchParams;
      const from = params.get('from');
      const to = params.get('to');
      if (!isValidDate(from) || !isValidDate(to)) return sendJson(res, 400, { error: 'Parameter from & to wajib YYYY-MM-DD.' });
      const items = db.prepare(`SELECT p.id AS product_id, p.name, SUM(s.qty) AS qty, SUM(s.total) AS total
        FROM sales s JOIN products p ON p.id = s.id
        WHERE date(s.created_at, '+7 hours') >= ? AND date(s.created_at, '+7 hours') <= ?
        GROUP BY p.id ORDER BY total DESC`).all(from, to);
      return sendJson(res, 200, { from, to, items, grand_total: items.reduce((sum, i) => sum + i.total, 0) });
    }
    if (req.method === 'GET' && !pathname.startsWith('/api/')) return await serveStatic(pathname, res);
    sendJson(res, 404, { error: 'Gak ketemu.' });
  } catch (err) {
    console.error(err);
    sendJson(res, 500, { error: 'Server error.' });
  }
});

const PORT = Number(process.env.PORT) || 3000;
server.listen(PORT, () => console.log(`Server jalan di port ${PORT}`));

=== FILE: public/app.js ===
const menu = document.querySelector('#menu');
const message = document.querySelector('#message');
const todayTotal = document.querySelector('#today-total');

const rupiah = (n) => `Rp${n.toLocaleString('id-ID')}`;

function say(text, isError = false) {
  message.textContent = text;
  message.classList.toggle('message--error', isError);
}

async function loadToday() {
  const res = await fetch('/api/sales/today');
  if (!res.ok) {
    say(`Gagal muat data: ${(await res.json()).error}`, true);
    return;
  }
  const { grand_total } = await res.json();
  todayTotal.textContent = rupiah(grand_total);
}

async function sell(product) {
  const res = await fetch('/api/sales', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ product_id: product.id, qty: 1 }),
  });
  if (!res.ok) return say((await res.json()).error, true);
  say(`${product.name} terjual.`);
  await Promise.all([loadMenu(), loadToday()]);
}

async function loadMenu() {
  const res = await fetch('/api/products');
  if (!res.ok) {
    say(`Gagal muat menu: ${(await res.json()).error}`, true);
    return;
  }
  const products = await res.json();
  menu.replaceChildren(...products.map((product) => {
    const button = document.createElement('button');
    button.type = 'button';
    const name = document.createElement('span');
    name.textContent = product.name;
    const price = document.createElement('span');
    price.textContent = rupiah(product.price);
    const stock = document.createElement('span');
    stock.textContent = product.low ? `Stok tipis: ${product.stock}` : `Stok ${product.stock}`;
    button.disabled = product.stock === 0;
    button.append(name, stock, price);
    button.addEventListener('click', () => sell(product));
    const item = document.createElement('li');
    item.append(button);
    return item;
  }));
}

try {
  await Promise.all([loadMenu(), loadToday()]);
} catch {
  say('Gak bisa nyambung ke server.', true);
}

## Review #1 — Laporan per menu & rename total -> grand_total (Fajar)

- **server.js, endpoint `/api/report`** — Fajar menulis `WHERE s.created_at >= ? AND s.created_at < ?` (filter mentah terhadap `created_at`, tanpa konversi zona waktu). Aturan: tanggal pakai WIB, dari dan to ikut dihitung. Ini memotong penjualan yang terjadi setelah jam 00:00 server (UTC+0) tapi masih hari yang diminta di WIB, plus membuat batas to jadi eksklusif. Dampak: omzet laporan bisa lebih kecil dari seharusnya, salah urut. Perbaikan: pakai `date(s.created_at, '+7 hours') >= ? AND date(s.created_at, '+7 hours') <= ?` (WIB, batas inklusif di kedua sisi).
- **server.js, endpoint `/api/report`** — tidak ada validasi input sama sekali. `from`/`to` bisa null (error SQL), string sembarang (SQL injection lewat string concatenation literal), atau `from` lebih besar dari `to` (hasil kosong tanpa tanda). Perbaikan: cek wajib ada, format `YYYY-MM-DD`, dan validasi tanggal benar secara kalender.
- **public/app.js, `loadToday()`** — Fajar ganti respons jadi `grand_total`, tapi frontend masih mengambil `{ total }`. Hasil: `todayTotal.textContent = rupiah(undefined)` → tampil "Rpundefined", total harian jadi tidak terbaca. Perbaikan: ambil `grand_total`.
- **public/app.js, loadToday & loadMenu** — tidak ada pengecekan `res.ok`. Pada error apa pun frontend tetap menulis `undefined` atau malah lempar exception. Perbaikan: periksa status, tampilkan pesan error dari body.
- **Server, `if (!body)`** — `readJson` mengembalikan `{}` jika body kosong, jadi pengecekan `if (!body)` tidak pernah true; permintaan tanpa body lolos. Bukan bug Fajar, tapi celah yang bisa dimanfaatkan saat tes. (Tidak diubah karena di luar scope diff Fajar dan menu utama tetap jalan.)
- **Database (migrasi)** — diff Fajar tidak menyentuh `db/schema.sql`, `002-stok.sql`, atau skema; `user_version` tetap 2, `sales` dan `products` tidak berubah. Tidak ada kolom NOT NULL baru tanpa DEFAULT. Aman.
