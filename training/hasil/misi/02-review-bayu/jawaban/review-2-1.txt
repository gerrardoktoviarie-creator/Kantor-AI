import { createServer } from 'node:http';
import { readFileSync } from 'node:fs';
import { readFile } from 'node:fs/promises';
import { extname, join, normalize, sep } from 'node:path';
import { fileURLToPath } from 'node:url';
import { DatabaseSync } from 'node:sqlite';

const ROOT = fileURLToPath(new URL('.', import.meta.url));
const PUBLIC = join(ROOT, 'public');

const db = new DatabaseSync(process.env.DB_PATH || join(ROOT, 'data.db'));
const fresh = !db.prepare("SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'products'").get();
if (fresh) db.exec(readFileSync(join(ROOT, 'db', 'schema.sql'), 'utf8'));
else if (db.prepare('PRAGMA user_version').get().user_version < 2) {
  db.exec(readFileSync(join(ROOT, 'db', 'migrations', '002-stok.sql'), 'utf8'));
}
db.exec('PRAGMA foreign_keys = ON');

const TYPES = { '.html': 'text/html; charset=utf-8', '.css': 'text/css', '.js': 'text/javascript' };
const PRODUCT = 'SELECT id, name, price, stock, min_stock, stock <= min_stock AS low FROM products';
const toProduct = (p) => p && { ...p, low: Boolean(p.low) };

function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify(data));
}

async function readJson(req) {
  let raw = '';
  for await (const chunk of req) raw += chunk;
  try { return raw ? JSON.parse(raw) : {}; } catch { return null; }
}

async function serveStatic(pathname, res) {
  const file = normalize(join(PUBLIC, pathname === '/' ? 'index.html' : pathname));
  if (!file.startsWith(PUBLIC + sep)) return sendJson(res, 404, { error: 'Gak ketemu.' });
  try {
    const body = await readFile(file);
    res.writeHead(200, { 'Content-Type': TYPES[extname(file)] ?? 'application/octet-stream' });
    res.end(body);
  } catch {
    sendJson(res, 404, { error: 'Gak ketemu.' });
  }
}

function todayInWib() {
  return db.prepare(`SELECT strftime('%Y-%m-%d', date('now', '+7 hours'), 'start of day') AS d`).get().d;
}

const server = createServer(async (req, res) => {
  const { pathname } = new URL(req.url, 'http://localhost');
  try {
    if (req.method === 'GET' && pathname === '/api/products') {
      return sendJson(res, 200, db.prepare(`${PRODUCT} ORDER BY name`).all().map(toProduct));
    }
    const restock = pathname.match(/^\/api\/products\/(\d+)\/restock$/);
    if (req.method === 'POST' && restock) {
      const body = await readJson(req);
      const qty = body?.qty;
      if (!Number.isInteger(qty) || qty < 1 || qty > 1000) return sendJson(res, 400, { error: 'qty harus 1-1000.' });
      const { changes } = db.prepare('UPDATE products SET stock = stock + ? WHERE id = ?').run(qty, Number(restock[1]));
      if (!changes) return sendJson(res, 404, { error: 'Menu gak ketemu.' });
      return sendJson(res, 200, toProduct(db.prepare(`${PRODUCT} WHERE id = ?`).get(Number(restock[1]))));
    }
    if (req.method === 'POST' && pathname === '/api/sales') {
      const body = await readJson(req);
      if (!body) return sendJson(res, 400, { error: 'Body harus JSON.' });
      const { product_id: productId, qty } = body;
      if (!Number.isInteger(qty) || qty < 1 || qty > 99) return sendJson(res, 400, { error: 'Jumlah harus 1-99.' });
      const product = db.prepare('SELECT id, price, stock FROM products WHERE id = ?').get(productId ?? 0);
      if (!product) return sendJson(res, 404, { error: 'Menu gak ketemu.' });
      db.exec('BEGIN');
      try {
        const { changes } = db.prepare('UPDATE products SET stock = stock - ? WHERE id = ? AND stock >= ?').run(qty, product.id, qty);
        if (!changes) {
          db.exec('ROLLBACK');
          return sendJson(res, 409, { error: `Stok gak cukup, sisa ${product.stock}.` });
        }
        const total = product.price * qty;
        const { lastInsertRowid } = db.prepare('INSERT INTO sales (product_id, qty, total) VALUES (?, ?, ?)').run(product.id, qty, total);
        db.exec('COMMIT');
        return sendJson(res, 201, { id: Number(lastInsertRowid), product_id: product.id, qty, total });
      } catch (err) {
        db.exec('ROLLBACK');
        throw err;
      }
    }
    const cancel = pathname.match(/^\/api\/sales\/(\d+)$/);
    if (req.method === 'DELETE' && cancel) {
      const sale = db.prepare(`SELECT id, product_id, qty
        FROM sales WHERE id = ?`).get(Number(cancel[1]));
      if (!sale) return sendJson(res, 404, { error: 'Penjualan gak ketemu.' });
      const { d } = db.prepare(`SELECT date('now', '+7 hours') AS d`).get();
      if (db.prepare(`SELECT 1 FROM sales WHERE id = ? AND date(created_at, '+7 hours') = ?`).get(sale.id, d)) {
        db.exec('BEGIN');
        db.prepare('DELETE FROM sales WHERE id = ?').run(sale.id);
        db.prepare('UPDATE products SET stock = stock + ? WHERE id = ?').run(sale.qty, sale.product_id);
        db.exec('COMMIT');
        return sendJson(res, 200, { ok: true });
      }
      return sendJson(res, 409, { error: 'Cuma penjualan hari ini yang bisa dibatalin.' });
    }
    if (req.method === 'GET' && pathname === '/api/sales/today') {
      const sales = db.prepare(`SELECT s.id, s.product_id, p.name, s.qty, s.total, s.created_at
        FROM sales s JOIN products p ON p.id = s.product_id
        WHERE date(s.created_at, '+7 hours') = date('now', '+7 hours') ORDER BY s.id DESC`).all();
      return sendJson(res, 200, { sales, total: sales.reduce((sum, s) => sum + s.total, 0) });
    }
    // Laporan per menu. from/to: YYYY-MM-DD (WIB), dua-duanya ikut dihitung.
    if (req.method === 'GET' && pathname === '/api/report') {
      const params = new URL(req.url, 'http://localhost').searchParams;
      const from = params.get('from') ?? '';
      const to = params.get('to') ?? '';
      const isDate = (d) => /^(\d{4})-(\d{2})-(\d{2})$/.test(d)
        && d >= '0000-01-01' && d <= '9999-12-31'
        && new Date(`${d}T00:00:00Z`).toISOString().startsWith(d);
      const valid = isDate(from) && isDate(to) && from <= to;
      if (!valid) return sendJson(res, 400, { error: 'from dan to harus tanggal YYYY-MM-DD, from <= to.' });
      const items = db.prepare(`SELECT p.id AS product_id, p.name, SUM(s.qty) AS qty, SUM(s.total) AS total
        FROM sales s JOIN products p ON p.id = s.product_id
        WHERE date(s.created_at, '+7 hours') >= date(?) AND date(s.created_at, '+7 hours') <= date(?)
        GROUP BY p.id ORDER BY total DESC`).all(from, to);
      return sendJson(res, 200, { from, to, items, grand_total: items.reduce((sum, i) => sum + i.total, 0) });
    }
    if (req.method === 'GET' && !pathname.startsWith('/api/')) return await serveStatic(pathname, res);
    sendJson(res, 404, { error: 'Gak ketemu.' });
  } catch (err) {
    console.error(err);
    sendJson(res, 500, { error: 'Server error.' });
  }
});

const PORT = Number(process.env.PORT) || 3000;
server.listen(PORT, () => console.log(`Server jalan di port ${PORT}`));
