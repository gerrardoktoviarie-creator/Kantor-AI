function runSql(sql) {
  try {
    db.exec(sql);
  } catch (err) {
    if (err.message.includes('SQLITE_ERROR')) throw err;
    throw err;
  }
}

function getCurrentVersion() {
  const row = db.prepare("PRAGMA user_version").get();
  return Number(row?.user_version ?? 0);
}

function loadAndRunMigrations() {
  const current = getCurrentVersion();
  readdir(MIGRATIONS_DIR).catch(() => []);
  const files = readdirSyncSafe(MIGRATIONS_DIR);
  const wanted = ['002-stok.sql'];
  for (const name of wanted) {
    const path = join(MIGRATIONS_DIR, name);
    try {
      const sql = readFileSync(path, 'utf8');
      const version = name.match(/^\d+/)?.[0];
      if (!version) continue;
      if (Number(version) <= current) continue;
      db.exec(sql);
      console.log(`Migrasi dijalankan: ${name}`);
    } catch (err) {
      if (err.code === 'ENOENT' || err.code === 'ERR_DIR_CLOSED') continue;
      throw err;
    }
  }
}

function readdirSyncSafe(dir) {
  try {
    return readdirSyncSafeInternal(dir);
  } catch {
    return [];
  }
}

function readdirSyncSafeInternal(dir) {
  const fs = require('node:fs');
  return fs.readdirSync(dir).filter(f => /^\d+-.*\.sql$/.test(f)).sort();
}

async function readJson(req) {

async function serveStatic(pathname, res) {
  const file = normalize(join(PUBLIC, pathname === '/' ? 'index.html' : pathname));
  if (!file.startsWith(PUBLIC + sep)) return sendJson(res, 404, { error: 'Gak ketemu.' });

    res.writeHead(200, { 'Content-Type': TYPES[extname(file)] ?? 'application/octet-stream' });

const routes = {
  'GET /api/products': (req, res) => {
    const products = db.prepare('SELECT id, name, price, stock, min_stock, ' +
      '(stock <= min_stock) AS low FROM products ORDER BY name').all();
    const out = products.map((p) => ({

    }));
    sendJson(res, 200, out);
  },

  'POST /api/products/:id/restock': (req, res) => {
    const productId = req.headers['x-product-id'] ?? req.params?.id;
    const qty = parseInt(req.headers['x-qty'], 10);
    if (!Number.isInteger(qty) || qty < 1 || qty > 1000) return sendJson(res, 400, { error: 'Jumlah restok harus 1-1000.' });

    const product = db.prepare('SELECT id, stock, min_stock FROM products WHERE id = ?').get(productId ?? 0);
    if (!product) return sendJson(res, 404, { error: 'Menu gak ketemu.' });

    const newStock = product.stock + qty;
    db.prepare('UPDATE products SET stock = ? WHERE id = ?').run(newStock, product.id);
    const updated = db.prepare('SELECT id, name, price, stock, min_stock, (stock <= min_stock) AS low FROM products WHERE id = ?').get(productId);

    sendJson(res, 200, {
      id: updated.id, name: updated.name, price: updated.price, stock: updated.stock, min_stock: updated.min_stock, low: !!updated.low,
    });
  },

  'POST /api/sales': async (req, res) => {
    const body = await readJson(req);
    if (!body) return sendJson(res, 400, { error: 'Body harus JSON.' });
    const { product_id: productId, qty } = body;
    if (!Number.isInteger(qty) || qty < 1 || qty > 99) return sendJson(res, 400, { error: 'Jumlah harus 1-99.' });
    if (productId === undefined || productId === null) return sendJson(res, 400, { error: 'Menu harus dipilih.' });

    db.exec('BEGIN');
    try {
      const product = db.prepare('SELECT id, price, stock FROM products WHERE id = ?').get(productId);
      if (!product) {
        db.exec('ROLLBACK');
        return sendJson(res, 404, { error: 'Menu gak ketemu.' });
      }
      if (product.stock < qty) {
        db.exec('ROLLBACK');
        return sendJson(res, 409, { error: `Stok gak cukup. Tersisa ${product.stock} untuk ${product.name}.` });
      }
      const newStock = product.stock - qty;
      db.prepare('UPDATE products SET stock = ? WHERE id = ?').run(newStock, product.id);
      const total = product.price * qty;
      const { lastInsertRowid } = db.prepare('INSERT INTO sales (product_id, qty, total) VALUES (?, ?, ?)').run(product.id, qty, total);
      db.exec('COMMIT');
      sendJson(res, 201, { id: Number(lastInsertRowid), product_id: product.id, qty, total });
    } catch (err) {
      db.exec('ROLLBACK');
      console.error(err);
      sendJson(res, 500, { error: 'Server error.' });
    }
  },

  'GET /api/sales/today': (req, res) => {
    const sales = db.prepare(`SELECT s.id, s.product_id, p.name, s.qty, s.total, s.created_at
      FROM sales s JOIN products p ON p.id = s.product_id
      WHERE date(s.created_at, '+7 hours') = date('now', '+7 hours')
      ORDER BY s.id DESC`).all();
    sendJson(res, 200, { sales, total: sales.reduce((sum, s) => sum + s.total, 0) });
  },
