import { readFileSync } from 'node:fs';
import { join, normalize, sep } from 'node:path';
import { fileURLToPath } from 'node:url';
import { DatabaseSync } from 'node:sqlite';

const ROOT = fileURLToPath(new URL('.', import.meta.url));
const db = new DatabaseSync(process.env.DB_PATH || join(ROOT, 'data.db'));

if (db.prepare('PRAGMA user_version').get().user_version < 5) {
  const hasCategory = db.prepare("SELECT name FROM pragma_table_info('products') WHERE name = 'category'").get();
  if (!hasCategory) {
    // Tabel lama mungkin punya kategori NOT NULL; alihkan jadi nullable dulu biar UPDATE jalan.
    try {
      db.exec("ALTER TABLE products ALTER COLUMN category DROP NOT NULL");
    } catch {
      // SQLite dialect lama / kolom udah nullable: abaikan.
    }
    db.exec(`ALTER TABLE products ADD COLUMN category TEXT CHECK (category IN ('minuman', 'makanan'))`);
    db.prepare('UPDATE products SET category = "makanan" WHERE name = "Roti Bakar Cokelat"').run();
    db.prepare('UPDATE products SET category = "minuman" WHERE name != "Roti Bakar Cokelat"').run();
  }
  db.prepare('PRAGMA user_version = 5').run();
}

db.close();

PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS products (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL UNIQUE,
  price INTEGER NOT NULL CHECK (price >= 0),
  stock INTEGER NOT NULL DEFAULT 0 CHECK (stock >= 0),
  min_stock INTEGER NOT NULL DEFAULT 5 CHECK (min_stock >= 0),
  category TEXT CHECK (category IN ('minuman', 'makanan')),
  created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS sales (
  id INTEGER PRIMARY KEY,
  product_id INTEGER NOT NULL REFERENCES products(id),
  qty INTEGER NOT NULL CHECK (qty > 0),
  total INTEGER NOT NULL CHECK (total >= 0),
  created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE INDEX IF NOT EXISTS sales_created ON sales (created_at);

## Review #3 — Migrasi kategori (Joko)

1. **db/migrations/005-pisah-perubahan-kategori.js — syntax error fatal saat startup.** `db.exec()` menerima SQL, bukan JS, jadi `import { readFileSync } ...` langsung error "syntax error" dan server mati di semua kondisi (versi 1, 2, baru, versi 3, fitur lama). Dampak: kasir tidak bisa jualan. Perbaikan: script migrasi sekarang kode JS murni (bukan string JS), membaca schema, mengecek keberadaan kolom `category`, menambahkannya tanpa memaksa NOT NULL, mengassign nilai per nama menu, lalu menaikkan versi.

2. **Server.js hanya panggil migrasi 005, lewatkan 004.** Versi server saat ini (`if (version < 5)`) menjalankan JS tapi melompati SQL `004-kategori-nullable.sql`. Di laptop kasir (versi 2) migrasi 003 sempat membuat kategori NOT NULL, sehingga langkah ADD COLUMN tanpa DEFAULT/nilai akan gagal. Perbaikan: tambahkan urutan eksekusi `version < 4` untuk 004 dulu (DROP NOT NULL), baru `version < 5` untuk 005. Script 005 juga sudah dilindungi try-catch untuk kasus kolom udah nullable/dialect beda.

3. **db/schema.sql — instalasi baru jadi salah versi.** Schema bawaan masih `user_version = 3` (versi 003), padahal migrasi 004/005 sudah ada. Dampak: install baru langsung punya kategori (benar), tapi jalur upgrade dari 3 ke 5 terlewat dan migrasi 004 tidak pernah dijalankan pada fresh install. Perbaikan: naikkan `PRAGMA user_version = 5` di schema, sehingga fresh install langsung setara versi terakhir dan skema (termasuk kolom `category` opsional) sinkron.

4. **db/migrations/003-kategori.sql — tetap disimpan sebagai rekam jejak.** Sudah direvisi sebelumnya: `NOT NULL` dilepas, update per nama menu. Tetap dipakai saat upgrade dari versi <3 ke 3.

Tolong jalankan: hapus data.db → `node server.js` (harusnya kategori langsung terisi minuman/makanan); lalu `node server.js` berulang kali (harus aman, tidak error "duplicate column").

- Tidak ada perubahan skema database (kolom `category` opsional, tanpa NOT NULL), jadi tidak butuh backup. Server restart otomatis menjalankan upgrade bertahap; bisa diulang berkali-kali tanpa merusak.
